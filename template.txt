local function FindDreamItem(player)
    if player == nil or player.components == nil or player.components.inventory == nil then
        return nil
    end

    local inv = player.components.inventory
    local hand_item = inv:GetEquippedItem(EQUIPSLOTS.HANDS)
    if hand_item ~= nil and hand_item.prefab == "dream" then
        return hand_item
    end

    for _, item in pairs(inv.itemslots) do
        if item ~= nil and item.prefab == "dream" then
            return item
        end
    end

    return nil
end

local function FindEquippedDreamItem(player)
    if player == nil or player.components == nil or player.components.inventory == nil then
        return nil
    end

    local hand_item = player.components.inventory:GetEquippedItem(EQUIPSLOTS.HANDS)
    return hand_item ~= nil and hand_item.prefab == "dream" and hand_item or nil
end

local DREAM_AUTO_PICKUP_RADIUS = 4
local DREAM_AUTO_PICKUP_INTERVAL = 0.25
local DREAM_STAR_MAX_DISTANCE = 30
local DREAM_PAN_FLUTE_RANGE = 15
local DREAM_PAN_FLUTE_DURATION = 20
local DREAM_AUTO_PICKUP_MUST_TAGS = { "_inventoryitem" }
local DREAM_AUTO_PICKUP_CANT_TAGS = { "INLIMBO", "NOCLICK", "catchable", "fire", "minesprung" }

local function CanDreamAutoPickupItem(item)
    if item == nil or not item:IsValid() or item.components == nil or item.components.inventoryitem == nil then
        return false
    end

    local inventoryitem = item.components.inventoryitem
    if not inventoryitem.canbepickedup then
        return false
    end

    return inventoryitem.IsHeld == nil or not inventoryitem:IsHeld()
end

local function GiveItemToDream(player, dream_item, item)
    local position = item:GetPosition()
    if player.components.inventory ~= nil then
        return player.components.inventory:GiveItem(item, nil, position, true)
    end

    return false
end

local StopDreamAutoPickup

local function DreamAutoPickupTick(player)
    if player == nil or not player:IsValid() or player.components == nil or player.components.inventory == nil then
        return
    end

    local dream_item = FindEquippedDreamItem(player)
    if dream_item == nil then
        StopDreamAutoPickup(player)
        return
    end

    local x, y, z = player.Transform:GetWorldPosition()
    local items = GLOBAL.TheSim:FindEntities(x, y, z, DREAM_AUTO_PICKUP_RADIUS, DREAM_AUTO_PICKUP_MUST_TAGS, DREAM_AUTO_PICKUP_CANT_TAGS)
    for _, item in ipairs(items) do
        if CanDreamAutoPickupItem(item) then
            if GiveItemToDream(player, dream_item, item) then
                break
            end
        end
    end
end

StopDreamAutoPickup = function(player)
    if player._dream_auto_pickup_task ~= nil then
        player._dream_auto_pickup_task:Cancel()
        player._dream_auto_pickup_task = nil
    end
    player._dream_auto_pickup_enabled = nil
end

local function StartDreamAutoPickup(player)
    StopDreamAutoPickup(player)
    player._dream_auto_pickup_enabled = true
    player._dream_auto_pickup_task = player:DoPeriodicTask(DREAM_AUTO_PICKUP_INTERVAL, DreamAutoPickupTick)
    DreamAutoPickupTick(player)
end

local function ToggleDreamAutoPickup(player)
    if FindEquippedDreamItem(player) == nil then
        return
    end

    if player._dream_auto_pickup_enabled then
        StopDreamAutoPickup(player)
    else
        StartDreamAutoPickup(player)
    end

    if player.components.talker ~= nil then
        player.components.talker:Say(player._dream_auto_pickup_enabled and "Dream auto pickup: ON" or "Dream auto pickup: OFF")
    end
end

local function CallDreamStar(player, x, z)
    if FindEquippedDreamItem(player) == nil then
        return
    end

    local px, py, pz = player.Transform:GetWorldPosition()
    if type(x) ~= "number" or type(z) ~= "number" then
        x, z = px, pz
    end

    local dx = x - px
    local dz = z - pz
    if dx * dx + dz * dz > DREAM_STAR_MAX_DISTANCE * DREAM_STAR_MAX_DISTANCE then
        x, z = px, pz
    end

    if GLOBAL.TheWorld.Map ~= nil and not GLOBAL.TheWorld.Map:IsPassableAtPoint(x, py, z) then
        x, z = px, pz
    end

    local star = GLOBAL.SpawnPrefab("stafflight")
    if star ~= nil then
        star.Transform:SetPosition(x, 0, z)
    end
end

local function PlayDreamPanFlute(player)
    if FindEquippedDreamItem(player) == nil then
        return
    end

    local px, py, pz = player.Transform:GetWorldPosition()

    local targets = GLOBAL.TheSim:FindEntities(
        px,
        py,
        pz,
        DREAM_PAN_FLUTE_RANGE,
        { "_combat" },
        { "INLIMBO", "player", "playerghost", "wall", "structure", "epic" }
    )

    for _, target in ipairs(targets) do
        if target ~= player
            and target.components ~= nil
            and target.components.sleeper ~= nil
            and target.components.sleeper:IsAsleep() == false then
            target.components.sleeper:GoToSleep(DREAM_PAN_FLUTE_DURATION)
        end
    end

    if player.SoundEmitter ~= nil then
        player.SoundEmitter:PlaySound("dontstarve/common/pan_flute")
    end

    if player.components.talker ~= nil then
        player.components.talker:Say("Pan Flute")
    end
end

    -- Tỷ lệ kích hoạt hiệu ứng (0.25 tức là 25% cơ hội xuất hiện hiệu ứng khi đánh)
    local PROBABILITY = 0.25 

    -- Hàm xử lý các hiệu ứng ngẫu nhiên
    local function ApplyRandomAttackEffect(attacker, data)
        -- Đảm bảo có mục tiêu và mục tiêu vẫn còn sống/hợp lệ
        local target = data and data.target
        if not target or not target:IsValid() then return end

        -- Thử vận may: Nếu số ngẫu nhiên lớn hơn tỷ lệ đặt ra thì bỏ qua
        if GLOBAL.math.random() > PROBABILITY then return end

        -- Chọn ngẫu nhiên 1 trong 4 hiệu ứng (từ 1 đến 4)
        local effect_type = GLOBAL.math.random(1, 4)

        if effect_type == 1 then
            -----------------------------------------------------------
            -- ⚡ HIỆU ỨNG 1: GIẬT ĐIỆN (SÉT ĐÁNH)
            -----------------------------------------------------------
            local lightning = GLOBAL.SpawnPrefab("lightning")
            if lightning then
                lightning.Transform:SetPosition(target.Transform:GetWorldPosition())
                -- Gây thêm sát thương điện nếu muốn (tùy chọn)
                if target.components.combat then
                    target.components.combat:GetAttacked(attacker, 20) -- Gây thêm 20 sát thương
                end
            end
            if attacker.components.talker then
                attacker.components.talker:Say("⚡ Giật điện!")
            end

        elseif effect_type == 2 then
            -----------------------------------------------------------
            -- ❄️ HIỆU ỨNG 2: ĐÓNG BĂNG MỤC TIÊU
            -----------------------------------------------------------
            if target.components.freezable then
                -- Thêm độ lạnh vào mục tiêu, số càng cao đóng băng càng nhanh/lâu
                target.components.freezable:AddColdness(2) 
                target.components.freezable:SpawnShatterFX()
            end
            if attacker.components.talker then
                attacker.components.talker:Say("❄️ Đóng băng!")
            end

        elseif effect_type == 3 then
            -----------------------------------------------------------
            -- 🔥 HIỆU ỨNG 3: THIÊU CHÁY
            -----------------------------------------------------------
            if target.components.burnable and not target.components.burnable:IsBurning() then
                target.components.burnable:Ignite()
            end
            if attacker.components.talker then
                attacker.components.talker:Say("🔥 Thiêu cháy!")
            end

        elseif effect_type == 4 then
            -----------------------------------------------------------
            -- 🩸 HIỆU ỨNG 4: HÚT MÁU (HỒI MÁU CHO NGƯỜI CHƠI)
            -----------------------------------------------------------
            if attacker.components.health and not attacker.components.health:IsDead() then
                local heal_amount = 10 -- Số máu hồi lại cho người chơi
                attacker.components.health:DoDelta(heal_amount)
                
                -- Tạo hiệu ứng hạt màu đỏ của dơi hút máu (Batilisk) để nhìn trực quan hơn
                local fx = GLOBAL.SpawnPrefab("vampire_bat_fx")
                if fx then
                    fx.Transform:SetPosition(attacker.Transform:GetWorldPosition())
                end
            end
            if attacker.components.talker then
                attacker.components.talker:Say("🩸 Hút máu! (+10)")
            end
        end
    end

    -- Đăng ký lắng nghe sự kiện "onattackother" cho tất cả người chơi tham gia vào server
    AddPlayerPostInit(function(inst)
        inst:ListenForEvent("onattackother", ApplyRandomAttackEffect)
    end)

end

AddStategraphPostInit("wilson", function(sg)
    if ACTIONS == nil or ACTIONS.ATTACK == nil then
        return
    end
    local attack_handler = sg.actionhandlers ~= nil and sg.actionhandlers[ACTIONS.ATTACK] or nil
    if attack_handler == nil or attack_handler.deststate == nil then
        return
    end

    local old_attack_handler = attack_handler.deststate
    attack_handler.deststate = function(inst, action)
        local weapon = inst.components.inventory:GetEquippedItem(EQUIPSLOTS.HANDS)
        if weapon and weapon.prefab == "dream" then
            local attack_period = TUNING.DREAM_ATTACK_PERIOD or 0.5
            if attack_period <= 0 then
                attack_period = 0.05
            end
            local mult = 0.5 / attack_period
            inst.AnimState:SetDeltaTimeMultiplier(mult)
			
			inst:ListenForEvent("newstate", function(i, data)
                if data and data.statename ~= "attack" then
                    i.AnimState:SetDeltaTimeMultiplier(1.0)
                end
            end)
			
            return "attack"
        else
            inst.AnimState:SetDeltaTimeMultiplier(1.0)
            return old_attack_handler(inst, action)
        end
    end
end)

local function OnEquip(inst, owner)
    owner.AnimState:OverrideSymbol("swap_object", "swap_dream", "swap_dream")
    owner.AnimState:Show("ARM_carry")
    owner.AnimState:Hide("ARM_normal")
    if owner.components.combat ~= nil then
        inst._old_attack_period = owner.components.combat.min_attack_period

        local attack_period = TUNING.DREAM_ATTACK_PERIOD or 0.5
        owner.components.combat.min_attack_period = attack_period
        if owner.components.combat.ResetCooldown ~= nil then
            owner.components.combat:ResetCooldown()
        end
    end
    TurnOn(inst,owner)
end
  
local function OnUnequip(inst, owner)
    owner.AnimState:Hide("ARM_carry")
    owner.AnimState:Show("ARM_normal")
    if owner._dream_auto_pickup_task ~= nil then
        owner._dream_auto_pickup_task:Cancel()
        owner._dream_auto_pickup_task = nil
        owner._dream_auto_pickup_enabled = nil
    end
	if owner.components.combat ~= nil and inst._old_attack_period ~= nil then
		owner.components.combat.min_attack_period = inst._old_attack_period
		inst._old_attack_period = nil
	end
	TurnOff(inst,owner)
end

    inst.components.reticule.ease = true

    if not TheWorld.ismastersim then
        return inst
    end

    inst.fxcolour = {1, 145/255, 0}
    inst.castsound = "dontstarve/common/staffteleport"

    inst:AddComponent("blinkstaff")
    inst.components.blinkstaff:SetFX("sand_puff_large_front", "sand_puff_large_back")
    inst.components.blinkstaff.onblinkfn = onblink
     
    inst:AddComponent("weapon")
    inst.components.weapon:SetDamage(TUNING.DREAM_DAMAGE)
    inst.components.weapon:SetRange(20, 22)
    inst.components.weapon:SetProjectile("fire_projectile")

    -- Crit system
    inst.crit_chance = TUNING.DREAM_CRIT_CHANCE or 0.25
    inst.crit_multiplier = TUNING.DREAM_CRIT_MULTIPLIER or 1.5
    inst.projectile_chance = TUNING.DREAM_PROJECTILE_CHANCE or 0.3
    
    -- Override weapon hit to add crit damage and projectiles
    local old_onhitfn = inst.components.weapon.onhitfn
    inst.components.weapon.onhitfn = function(weapon, attacker, target)
        if old_onhitfn ~= nil then
            old_onhitfn(weapon, attacker, target)
        end
        
        -- Crit check
        local is_crit = math.random() < (weapon.crit_chance or 0.25)
        if is_crit and target ~= nil and target.components.combat ~= nil then
            local crit_damage = weapon.components.weapon:GetDamage() * ((weapon.crit_multiplier or 1.5) - 1)
            target.components.combat:GetAttacked(attacker, crit_damage, weapon, "none")
        end
        
        -- Projectile spawn
        if math.random() < (weapon.projectile_chance or 0.3) and attacker ~= nil and target ~= nil then
            local projectile_type = math.random() < 0.5 and "fire_projectile" or "ice_projectile"
            local proj = SpawnPrefab(projectile_type)
            if proj ~= nil then
                proj.Transform:SetPosition(attacker.Transform:GetWorldPosition())
                local dir = target:GetPosition() - attacker:GetPosition()
                local dist = dir:Length()
                if dist > 0 then
                    dir = dir:Normalize()
                    proj.components.projectile:Launch(dir * 10, nil, attacker)
                end
            end
        end
    end

    inst:AddComponent("tool")
    inst.components.tool:SetAction(ACTIONS.CHOP, 10)
    inst.components.tool:SetAction(ACTIONS.MINE, 10)
    inst.components.tool:SetAction(ACTIONS.HAMMER, 2.50)
    inst.components.tool:SetAction(ACTIONS.NET)
    inst.components.tool:SetAction(ACTIONS.DIG)
    -- inst.components.tool:SetAction(ACTIONS.SHAVE)

    -- inst.components.tool:SetAction(ACTIONS.FISH)
    -- inst.components.tool:SetAction(ACTIONS.TILL)
    -- inst.components.tool:SetAction(ACTIONS.POUR_WATER)
    -- inst.components.tool:SetAction(ACTIONS.REPAIR)
    inst.components.tool:EnableToughWork(true)
    inst:AddComponent("oar")
    inst.components.oar.force = 5
    inst.components.oar.max_velocity = 20

	  
    inst:AddComponent("inspectable")
      
    inst:AddComponent("inventoryitem")
	inst.components.inventoryitem.keepondeath = true
